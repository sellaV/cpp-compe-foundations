cmake_minimum_required(VERSION 3.16)
project(text_report LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(USE_REFERENCE "Build the reference instead of the learner implementation" OFF)
set(ANALYZER_SOURCE analyzer.cpp)
if(USE_REFERENCE)
    set(ANALYZER_SOURCE reference/analyzer.cpp)
endif()
add_library(analyzer ${ANALYZER_SOURCE})
target_include_directories(analyzer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(text_report main.cpp)
target_link_libraries(text_report PRIVATE analyzer)
add_executable(report_tests test.cpp)
target_link_libraries(report_tests PRIVATE analyzer)
enable_testing()
add_test(NAME report_unit_tests COMMAND report_tests)
